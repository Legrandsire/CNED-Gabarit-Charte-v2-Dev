%%% MACRO SUBTITLE
% Macro pour souligner uniquement le texte du titre
\NewDocumentCommand{\underlinetitle}{O{2pt} O{0.25ex} m}{%
  \tikz[baseline=(text.base)]{
    % On pose d'abord le texte sur la couche principale-
    \node[inner sep=0pt] (text) {#3};
    % Puis on trace le trait en-dessous, sur une couche "derrière"
    \begin{pgfonlayer}{background}
      \draw[line width=#1, background2]
        ([yshift=-#2]text.base west) -- ([yshift=-#2]text.base east);
    \end{pgfonlayer}
  }%
}

\newcommand{\fixednum}[1]{\makebox[0.5em][l]{#1}}

\titleformat{\subsection}
  {\robotoCondensed\bfseries\fontsize{21}{24}\selectfont\RaggedRight}      % style du texte
  {\fixednum{\thesubsection}}         % numéro (normal)
  {1em}                    % espace numéro - titre
  {\bfseries\color{primarycolor}\underlinetitle}        % ici on applique la macro au texte seul
  []
\renewcommand{\thesubsection}{\Roman{subsection}.}
