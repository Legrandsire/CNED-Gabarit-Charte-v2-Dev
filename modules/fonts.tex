\usepackage{fontspec,titlesec,soul}

\renewcommand{\ULthickness}{10pt}

% --- 1) Italique spécial titres : robuste + compatible soul ---
\DeclareRobustCommand{\TitleTextIt}[1]{%
  {\addfontfeatures{FakeSlant=0.2}#1}%  % incline la fonte courante (même Cond Bold)
}
% Autoriser ces macros dans \ul{...}
\soulregister\TitleTextIt{1}
\soulregister\textit{1}
\soulregister\emph{1}
\soulregister\textbf{1}
\soulregister\textcolor{2}

% --- 2) Hooks pour (re)mapper \textit dans les titres ---
\newcommand{\TitleTextBegin}{%
  \begingroup
  \let\textit\TitleTextIt
  \let\emph\TitleTextIt
}
\newcommand{\TitleTextEnd}{\endgroup}
