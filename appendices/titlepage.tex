      {\robotoCondensed\fontsize{16}{19}\selectfont \sousTitre \par}
    };



    % Récupérer les coordonnées de la boite et création de points intermédiaires.
    %%% Cordonnées de la boite
    \coordinate (NW) at ([shift={(-\pgflinewidth+20,-\pgflinewidth+25)}]box.north west);
    \coordinate (NE) at ([shift={(\pgflinewidth+20,-\pgflinewidth+25)}]box.north east);
    \coordinate (SW) at ([shift={(-\pgflinewidth+20,\pgflinewidth+15)}]box.south west);
    \coordinate (SE) at ([shift={(\pgflinewidth+20,\pgflinewidth+15)}]box.south east);
    %%% Points intermédiaires

    \coordinate (N_color) at ($ (N_black)!0.5!(NE) $);

    % ========= L NOIR en POLYGONE (épaisseur 25pt) =========
    \pgfmathsetlengthmacro{\W}{25pt}   % épaisseur du L noir
    \pgfmathsetlengthmacro{\R}{0.5*\W} % demi-épaisseur (= rayon d’arrondi)

    % Contour du L (sens horaire) : points de base
    \coordinate (A) at ($(N_black)+(-\R,\R)$); % haut-gauche (extérieur)
    \coordinate (B) at ($(NE)+(\R,\R)$);       % haut-droit (extérieur)
    \coordinate (C) at ($(SE)+(\R,-\R)$);      % bas-droit (extérieur)
    \coordinate (D) at ($(SE)+(-\R,-\R)$);     % bas-gauche jambage (intérieur)
    \coordinate (E) at ($(NE)+(-\R,-\R)$);     % sous le coude (intérieur)
    \coordinate (F) at ($(N_black)+(-\R,-\R)$);% bas-gauche barre (extérieur)

    % L noir avec arrondis UNIQUEMENT en A (haut-gauche) et C (bas-droit)
    \path[fill=black]
      % --- A arrondi : de (F) vers (B) en passant par A ---
      ($(A)+(0,-\R)$)
        arc[start angle=180, end angle=90, radius=\R]
      -- (B)
      % descente sur le bord droit jusqu’à \R au-dessus de C
      -- ($(C)+(0,\R)$)
      % --- C arrondi ---
        arc[start angle=0, end angle=-90, radius=\R]
      -- (D) -- (E) -- (F) -- cycle;

    % ========= ORANGE : RECTANGLES (épaisseur 15pt) =========
    \pgfmathsetlengthmacro{\Wo}{15pt}
    \pgfmathsetlengthmacro{\Ro}{0.5*\Wo}

    % Rectangle orange vertical (mêmes offsets que tes \draw d'origine)

      ($(NE)+(-17.5pt-\Ro,-12.5pt)$) --
      ($(SE)+(-17.5pt-\Ro,-\R)$) --
      ($(SE)+(-17.5pt+\Ro,-\R)$) --
      ($(NE)+(-17.5pt+\Ro,-12.5pt)$) --
      cycle;

    % Rectangle orange horizontal (mêmes offsets que tes \draw d'origine)
    \path[fill=background1]
      ($(N_color)+(-17.5pt,-20pt-\Ro)$) --
      ($(NE)+(-17.5pt,-20pt-\Ro)$) --
      ($(NE)+(-17.5pt,-20pt+\Ro)$) --
      ($(N_color)+(-17.5pt,-20pt+\Ro)$) --
      cycle;

\end{tikzpicture}

    \node[
        rectangle,
        fill=none,

        inner xsep=3pt,
        inner ysep=3pt,
    ] (main) {\bfseries\fontsize{12}{15}\selectfont\getsubtitle};

    % Cadre avec UN seul coin arrondi
\begin{pgfonlayer}{background}
  \def\padx{5pt}
  \def\pady{5pt}
  \def\r{8pt} % rayon

  % Coins du rectangle extérieur (autour du node)
  \coordinate (SW) at ([xshift=-\padx,yshift=-\pady]main.south west);
  \coordinate (SE) at ([xshift= \padx,yshift=-\pady]main.south east);
  \coordinate (NE) at ([xshift= \padx,yshift= \pady]main.north east);
  \coordinate (NW) at ([xshift=-\padx,yshift= \pady]main.north west);

  \path[fill=primarycolor]
    (SW) --
    % bas : jusqu'à r avant SE
    ([xshift=-\r]SE)
      arc (270:360:\r) % arrondi Bas-Droit, finit sur le bord droit à +r
    --
    % droite : jusqu'à NE
    (NE) --
    % haut : jusqu'à r après NW
    ([xshift=\r]NW)
      arc (90:180:\r)  % arrondi Haut-Gauche, finit sur le bord gauche à -r
    --
    % gauche : retour à SW
    (SW) -- cycle;
\end{pgfonlayer}
\end{tikzpicture}
